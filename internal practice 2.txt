1. ITEM_MASTER

-- Table Creation
CREATE TABLE item_master (
    ID VARCHAR(10) PRIMARY KEY,
    NAME VARCHAR(50),
    TYPE VARCHAR(30),
    PRICE INT,
    QUALITY VARCHAR(20),
    ITEM_PACK_DATE DATE
);
